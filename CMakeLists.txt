cmake_minimum_required(VERSION 3.24)
project(BindOptiX VERSION 0.1.0 DESCRIPTION "Time, transport and biology layer for optical biosensors" LANGUAGES NONE)

option(BINDOPTIX_BUILD_CORE "Build the compiled module bindoptix._core" ON)
option(BINDOPTIX_ENABLE_CUDA "Build the CUDA backend (stage Z11)" OFF)
option(BINDOPTIX_STATIC_LIBSTDCXX "Link libstdc++ and libgcc statically into _core (GCC; for conda Pythons with an old libstdc++)" OFF)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT BINDOPTIX_BUILD_CORE)
  message(STATUS "BindOptiX: core disabled, pure Python reference only")
  return()
endif()
enable_language(CXX)

add_library(bindoptix_core STATIC cpp/src/cpu.cpp)
target_include_directories(bindoptix_core PUBLIC cpp/include)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(bindoptix_core PUBLIC OpenMP::OpenMP_CXX)
endif()
set_target_properties(bindoptix_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(bindoptix_core PRIVATE /W4)
else()
  # no -ffast-math: the core must reproduce the Python reference (NaN tests, rounding)
  target_compile_options(bindoptix_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BINDOPTIX_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86 89)  # RTX 3090, RTX 6000 Ada
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit 12.6 REQUIRED)
  add_library(bindoptix_cuda STATIC cpp/src/cuda.cu)
  target_include_directories(bindoptix_cuda PUBLIC cpp/include)
  target_compile_definitions(bindoptix_cuda PUBLIC BINDOPTIX_HAVE_CUDA)
  target_compile_definitions(bindoptix_core PUBLIC BINDOPTIX_HAVE_CUDA)
  # IEEE-exact device math so that results stay close to the CPU core and the reference
  target_compile_options(bindoptix_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(bindoptix_cuda PUBLIC CUDA::cudart_static)
  set_target_properties(bindoptix_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()

if(SKBUILD)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/python/bindings.cpp)
  target_link_libraries(_core PRIVATE bindoptix_core)
  if(BINDOPTIX_ENABLE_CUDA)
    target_link_libraries(_core PRIVATE bindoptix_cuda)
  endif()
  if(BINDOPTIX_STATIC_LIBSTDCXX)
    target_link_options(_core PRIVATE -static-libstdc++ -static-libgcc)
  endif()
  install(TARGETS _core LIBRARY DESTINATION bindoptix)
endif()

message(STATUS "BindOptiX ${PROJECT_VERSION}: core=${BINDOPTIX_BUILD_CORE} CUDA=${BINDOPTIX_ENABLE_CUDA}")
